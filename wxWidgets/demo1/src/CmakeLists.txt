cmake_minimum_required(VERSION 3.16)

project(demo1 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

set(wxWidgets_ROOT_DIR "D:/wxWeigets/wxWidgets-3.3.3")
set(wxWidgets_LIB_DIR "${wxWidgets_ROOT_DIR}/lib/vc_x64_lib")
find_package(wxWidgets 3.2 REQUIRED COMPONENTS core base)
include(${wxWidgets_USE_FILE})

add_executable(demo1 WIN32
    App.cpp
    MainFrame.cpp
)
target_include_directories(demo1 PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(demo1 PRIVATE ${wxWidgets_LIBRARIES})

if(MSVC)
    target_compile_options(demo1 PRIVATE /utf-8)
endif()
